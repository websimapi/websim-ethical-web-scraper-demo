// Sample code snippet
const { scrapeProductInfo } = require('./scraper');

app.get('/api/products', async (req, res) => {
  try {
    const { query } = req.query;
    const results = await scrapeProductInfo(query);
    res.json(results);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});
